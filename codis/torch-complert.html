def entrena():
    model.train()
    running_loss = 0.0
    correct = 0
    total = 0
    for inputs, targets in train_loader:
        inputs, targets = inputs.to(device), targets.to(device)
        optimizer.zero_grad()
        outputs = model(inputs)
#        outputs = outputs.squeeze()
        loss = criterion(outputs, targets)
        running_loss += loss.item() * inputs.size(0)
        loss.backward()
        optimizer.step()
        # Calcular accuracy
        # correct += ((outputs > 0.5).float()== targets).sum().item()
        _, preds = torch.max(outputs, 1)
        correct += (preds == targets).sum().item()
        total   += targets.size(0)
    return running_loss / total, correct / total

def avalua(loader):
    model.eval()
    running_loss = 0.0
    correct = 0
    total = 0
    predsTl = torch.tensor([]).to(device)
    with torch.no_grad():
        for inputs, targets in loader:
            inputs, targets = inputs.to(device), targets.to(device)
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            running_loss += loss.item() * inputs.size(0)
            # Calcular accuracy
            # correct += ((outputs > 0.5).float()== targets).sum().item()
            _, preds = torch.max(outputs, 1)
            correct += (preds == targets).sum().item()
            total   += targets.size(0)
            predsTl = torch.cat((predsTl, preds), 0)
        return running_loss / total, correct / total, predsTl

num_epochs   = 200
patience     = 10                 # epochs sense millora a la validació
best_val_loss = float('inf')
best_weights = copy.deepcopy(model.state_dict())
epochs_no_improve = 0
ckpt_path = "/kaggle/working/iris_best_three_sets.pth"
dades = {'training_loss': [], 'training_accuracy': [], 'validation_loss': [], 'validation_accuracy': []}
for epoch in range(num_epochs):
    ### training batch ###
    trn_loss, trn_acc = entrena()
    dades['training_loss'].append(trn_loss)
    dades['training_accuracy'].append(trn_acc)
    ### validation batch ###
    val_loss, val_acc, _ = avalua(val_loader)
    dades['validation_loss'].append(val_loss)
    dades['validation_accuracy'].append(val_acc)
    ### resultats batch ###
    print(f'Epoch: {epoch+1}, Loss: {trn_loss:.4f}, Acc: {trn_acc:.4f}, ValLoss: {val_loss:.4f}, ValAcc: {val_acc:.4f}')

    # ---------- Early‑stopping & checkpoint ----------
    if val_loss < best_val_loss:
        best_val_loss = val_loss
        best_weights = copy.deepcopy(model.state_dict())
        torch.save(best_weights, ckpt_path)
        epochs_no_improve = 0
        print("   -> New best model saved")
    else:
        epochs_no_improve += 1
        if epochs_no_improve >= patience:
            print(f"Early stopping triggered after {epoch} epochs.")
            break
