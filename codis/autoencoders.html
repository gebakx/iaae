epochs = 20
for epoch in range(epochs):
    autoencoder.train()
    optimizer.zero_grad()
    outputs = autoencoder(Xt)
    loss = criterion(outputs, Xt)
    loss.backward()
    optimizer.step()
    print(f'Epoch [{epoch+1}/{epochs}], ' +
          f'Loss: {loss.item():.4f}')
